# Text layer extraction
pypdf==5.1.0

# Page rendering for OCR. Also usable as a second text extractor, which is how
# the corpus corruption was confirmed to be in the files rather than the reader.
pymupdf==1.28.2

# Local Arabic and English OCR. Runs on this machine: the corpus is classified
# for internal use, so cloud OCR is not an option.
easyocr==1.7.2

# torch arrives as an easyocr dependency. On an NVIDIA GPU, install the CUDA
# build instead for roughly an 8x speedup:
#   pip install --force-reinstall torch torchvision \
#       --index-url https://download.pytorch.org/whl/cu126

# Embedding models for stage 6's bake-off, BAAI/bge-m3 against
# intfloat/multilingual-e5-large. Both load through this rather than
# sentence-transformers, so pooling is written out in embedder.py instead of
# hidden inside a wrapper: the one asymmetry that could silently decide the
# bake-off (e5 needs "query: "/"passage: " prefixes, BGE-M3 does not) has to
# stay visible in the source that runs it. Pinned to 4.46.3 rather than the
# 5.x release this project started on: the newer release's rewritten weight
# loading path was one of several suspects chased down while BGE-M3's
# pickle checkpoint crashed intermittently, and this older, mature release
# was kept once the real fix, process isolation, made the version change
# moot to verify either way.
transformers==4.46.3

# BGE-M3's sparse head, sparse_linear.pt, is not part of the dense encoder's
# own weights and is fetched with this directly, hf_hub_download, rather
# than through transformers' own loading path.
huggingface_hub==0.36.2

# Local vector store, file mode, no Docker. One collection per context
# variant.
qdrant-client==1.19.0

# Stage 7's device_probe.py reads this process's own RSS to measure the
# host memory cost of the CPU-versus-GPU query embedding trade-off. Already
# present as a transitive dependency of accelerate; declared directly here
# because this project imports it by name, not just benefits from it.
psutil==7.2.2
